cmake_minimum_required(VERSION 3.18)
project(scrcpy_plus_airplay LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(UX ${ROOT}/vendor/airplay/uxplay/lib)
set(PL ${ROOT}/vendor/airplay/libplist)
if(NOT NODE_INCLUDE_DIR)
  message(FATAL_ERROR "Pass -DNODE_INCLUDE_DIR=<Node headers directory>")
endif()
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
file(GLOB PL_SOURCES ${PL}/src/*.c ${PL}/libcnary/*.c)
add_library(plist STATIC ${PL_SOURCES})
target_include_directories(plist PUBLIC ${PL}/include PRIVATE ${PL}/src ${PL}/libcnary/include)
target_compile_definitions(plist PRIVATE PACKAGE_VERSION="source-snapshot")
include(CheckSymbolExists)
foreach(FEATURE STRNDUP STRPTIME LOCALTIME_R GMTIME_R)
  string(TOLOWER ${FEATURE} FUNCTION_NAME)
  check_symbol_exists(${FUNCTION_NAME} "string.h;time.h" HAVE_${FEATURE})
  if(HAVE_${FEATURE})
    target_compile_definitions(plist PRIVATE HAVE_${FEATURE})
  endif()
endforeach()
if(WIN32)
  target_compile_definitions(plist PRIVATE WIN32)
endif()
file(GLOB UX_SOURCES ${UX}/*.c ${UX}/playfair/*.c ${UX}/llhttp/*.c ${UX}/mdnsd/*.c)
add_library(airplay STATIC ${UX_SOURCES})
target_include_directories(airplay PUBLIC ${UX} PRIVATE ${UX}/playfair ${UX}/llhttp ${UX}/mdnsd)
target_compile_definitions(airplay PRIVATE PLIST_210 PLIST_230 OPENSSL_API_COMPAT=0x10101000L)
target_link_libraries(airplay PUBLIC plist OpenSSL::Crypto)
if(WIN32)
  # The distributed addon must not require a separate MinGW pthread runtime.
  find_library(WINPTHREAD_STATIC NAMES libwinpthread.a PATHS $ENV{MINGW_PREFIX}/lib REQUIRED)
  target_link_libraries(airplay PUBLIC ${WINPTHREAD_STATIC} ws2_32 iphlpapi crypt32 bcrypt)
else()
  target_link_libraries(airplay PUBLIC Threads::Threads)
endif()
add_library(plus_airplay MODULE bridge.cc)
target_include_directories(plus_airplay PRIVATE ${NODE_INCLUDE_DIR} ${ROOT}/node_modules/node-addon-api)
target_compile_definitions(plus_airplay PRIVATE NAPI_VERSION=8 NAPI_CPP_EXCEPTIONS)
target_link_libraries(plus_airplay PRIVATE airplay)
set_target_properties(plus_airplay PROPERTIES PREFIX "" SUFFIX ".node")
if(APPLE)
  execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE APPLE_SDK OUTPUT_STRIP_TRAILING_WHITESPACE)
  target_include_directories(plus_airplay SYSTEM PRIVATE ${APPLE_SDK}/usr/include/c++/v1)
  target_link_options(plus_airplay PRIVATE -undefined dynamic_lookup)
elseif(WIN32)
  target_link_options(plus_airplay PRIVATE -static-libgcc -static-libstdc++)
  target_link_libraries(plus_airplay PRIVATE ${NODE_LIBRARY})
endif()
install(TARGETS plus_airplay DESTINATION .)
